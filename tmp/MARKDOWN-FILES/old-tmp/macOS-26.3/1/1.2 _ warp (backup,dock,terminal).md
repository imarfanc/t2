## save current plists

- If you are pasting this into Warp/zsh directly, an even safer way is to run it as:

```sh
bash <<'EOF'
# paste script here
EOF
```

```bash
bash <<'ee'
#!/usr/bin/env bash

BACKUP_ROOT="$HOME/Developer/macos-reset"
STAMP="$(date +%Y-%m-%d_%H-%M-%S)"
OUT_DIR="$BACKUP_ROOT/$STAMP"
SUMMARY_FILE="$OUT_DIR/_summary.txt"

mkdir -p "$OUT_DIR"

backup_domain() {
  local label="$1"
  local domain="$2"
  local file="$3"

  if defaults read "$domain" > "$OUT_DIR/$file" 2>/dev/null; then
    printf "%-20s | OK      | %s\n" "$label" "$file" | tee -a "$SUMMARY_FILE"
  else
    printf "%-20s | FAILED  | %s\n" "$label" "$file" | tee -a "$SUMMARY_FILE"
    rm -f "$OUT_DIR/$file"
  fi
}

{
  echo "macOS settings backup"
  echo "Created: $(date)"
  echo "Folder: $OUT_DIR"
  echo
} > "$SUMMARY_FILE"

echo "========================================"
echo "macOS settings backup"
echo "========================================"
echo "Output folder: $OUT_DIR"
echo

backup_domain "Global"              NSGlobalDomain                 "global-defaults.txt"
backup_domain "Dock"                com.apple.dock                 "dock-settings.txt"
backup_domain "Finder"              com.apple.finder               "finder-settings.txt"
backup_domain "Safari"              com.apple.Safari               "safari-settings.txt"
backup_domain "Terminal"            com.apple.Terminal             "terminal-settings.txt"
backup_domain "TextEdit"            com.apple.TextEdit             "textedit-settings.txt"
backup_domain "Desktop Services"    com.apple.desktopservices      "desktopservices-settings.txt"

echo
echo "Done."
echo "Backup saved to: $OUT_DIR"
echo "Summary file:    $SUMMARY_FILE"
ee
```

| list all Preferences - `defaults domains | tr ',' '\n'`
| filter from all Preferences- `defaults domains | tr ',' '\n' | grep -i safari`

| delete a key in plist - `defaults delete -g AppleActionOnDoubleClick`
| view all settings - `defaults read -g`
| view dock settings - `defaults read com.apple.dock`

## dock

```sh
# remove dock icons
defaults write com.apple.dock persistent-apps -array
defaults write com.apple.dock persistent-others -array

# change dock settings
defaults write com.apple.dock orientation -string left
defaults write com.apple.dock autohide -bool true
defaults write com.apple.dock tilesize -integer 36
defaults write com.apple.dock magnification -bool true
defaults write com.apple.dock largesize -int 48
defaults write com.apple.dock show-recents -bool false

defaults write com.apple.dock persistent-apps -array-add '{tile-type="small-spacer-tile";}'
defaults write com.apple.dock persistent-apps -array-add '{tile-data={}; tile-type="spacer-tile";}'
defaults write com.apple.dock persistent-others -array-add '{tile-data={}; tile-type="spacer-tile";}'

# from macos-defaults.com
defaults write com.apple.dock "autohide-delay" -float "0"
defaults write com.apple.dock "autohide-time-modifier" -float "0"
defaults write com.apple.dock "scroll-to-open" -bool "true"

killall Dock
```
## terminal settings

Here's how to configure the "Clear Dark" profile with the following settings:
- Cursor type: line (2)
- Column count: 140
- Row count: 50
- Option key as Meta key: Enabled

```bash
#!/usr/bin/env bash

PLIST="$HOME/Library/Preferences/com.apple.Terminal.plist"
PROFILE="Window Settings:Clear Dark"

print_value() {
  local key_path="$1"
  /usr/libexec/PlistBuddy -c "Print '$key_path'" "$PLIST" 2>/dev/null || echo "<missing>"
}

set_or_add_with_report() {
  local short_key="$1"
  local type="$2"
  local value="$3"
  local full_key="$PROFILE:$short_key"

  local before after action
  before="$(print_value "$full_key")"

  if /usr/libexec/PlistBuddy -c "Set '$full_key' $value" "$PLIST" >/dev/null 2>&1; then
    action="Set"
  elif /usr/libexec/PlistBuddy -c "Add '$full_key' $type $value" "$PLIST" >/dev/null 2>&1; then
    action="Add"
  else
    action="FAILED"
  fi

  after="$(print_value "$full_key")"

  printf "%-20s | before: %-10s | action: %-6s | after: %s\n" \
    "$short_key" "$before" "$action" "$after"
}

echo "Checking profile..."
if ! /usr/libexec/PlistBuddy -c "Print '$PROFILE'" "$PLIST" >/dev/null 2>&1; then
  echo "Profile '$PROFILE' does not exist."
  echo "Create/copy it first, for example:"
  echo "/usr/libexec/PlistBuddy -c \"Copy 'Window Settings:Basic' '$PROFILE'\" \"$PLIST\""
  exit 1
fi

echo
echo "=== Before/After Updates for $PROFILE ==="
set_or_add_with_report "CursorType" integer 2
set_or_add_with_report "columnCount" integer 140
set_or_add_with_report "rowCount" integer 40
set_or_add_with_report "useOptionAsMetaKey" bool true
```

## some more settings (desktop & dock)

```sh
WM_PLIST=~/Library/Preferences/com.apple.WindowManager.plist
DOCK_PLIST=~/Library/Preferences/com.apple.dock.plist
#OUTDIR=~/Desktop
OUTDIR=~/Developer/macos-reset

mkdir -p "$OUTDIR" || exit 1

plutil -convert xml1 -o "$OUTDIR/windowmanager-before.xml" "$WM_PLIST" || exit 1
plutil -convert xml1 -o "$OUTDIR/dock-before.xml" "$DOCK_PLIST" || exit 1

defaults write com.apple.WindowManager EnableTiledWindowMargins -bool true
defaults write com.apple.dock enterMissionControlByTopWindowDrag -bool false
defaults write com.apple.dock expose-group-apps -bool true
defaults write com.apple.dock mru-spaces -bool false

killall Dock

sleep 0.5

plutil -convert xml1 -o "$OUTDIR/windowmanager-after.xml" "$WM_PLIST" || exit 1
plutil -convert xml1 -o "$OUTDIR/dock-after.xml" "$DOCK_PLIST" || exit 1

echo "=== WindowManager diff ==="
diff "$OUTDIR/windowmanager-before.xml" "$OUTDIR/windowmanager-after.xml"

echo
echo "=== Dock diff ==="
diff "$OUTDIR/dock-before.xml" "$OUTDIR/dock-after.xml"
```
