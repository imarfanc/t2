# warp (keyboard,trackpad)

## some macos settings

- If you are pasting this into Warp/zsh directly, an even safer way is to run it as:

```sh
bash <<'EOF'
# paste script here
EOF
```

```bash
#!/usr/bin/env bash

report_default() {
  local domain="$1"
  local key="$2"

  defaults read "$domain" "$key" 2>/dev/null || echo "<missing>"
}

write_with_report() {
  local label="$1"
  local domain="$2"
  local key="$3"
  local type="$4"
  local value="$5"

  local before after write_result
  before="$(report_default "$domain" "$key")"

  if defaults write "$domain" "$key" "$type" "$value" >/dev/null 2>&1; then
    write_result="OK"
  else
    write_result="FAILED"
  fi

  after="$(report_default "$domain" "$key")"

  printf "%-34s | before: %-15s | write: %-6s | after: %s\n" \
    "$label" "$before" "$write_result" "$after"
}

echo "========================================"
echo "macOS defaults update report"
echo "========================================"
echo

echo "=== Works now / no logout required ==="
write_with_report "Expand Save Dialog 1" NSGlobalDomain NSNavPanelExpandedStateForSaveMode -bool true
write_with_report "Expand Save Dialog 2" NSGlobalDomain NSNavPanelExpandedStateForSaveMode2 -bool true

write_with_report "Show Scroll Bars" NSGlobalDomain AppleShowScrollBars -string "WhenScrolling"
write_with_report "Scroller Paging Behavior" NSGlobalDomain AppleScrollerPagingBehavior -bool true

write_with_report "Tap to Click" NSGlobalDomain com.apple.mouse.tapBehavior -int 0

write_with_report "Auto Capitalization" NSGlobalDomain NSAutomaticCapitalizationEnabled -bool false
write_with_report "Press and Hold Accents" NSGlobalDomain ApplePressAndHoldEnabled -bool false

echo
echo "=== Usually needs logout / restart / new apps ==="
write_with_report "Key Repeat" NSGlobalDomain KeyRepeat -int 2
write_with_report "Initial Key Repeat" NSGlobalDomain InitialKeyRepeat -int 15
write_with_report "Keyboard Navigation" NSGlobalDomain AppleKeyboardUIMode -int 3
write_with_report "Function Keys Enabled" NSGlobalDomain com.apple.keyboard.fnState -bool true

echo
echo "Done."
echo
echo "Some changes may need one of these before they fully apply:"
echo "  - quit and reopen affected apps"
echo "  - log out and back in"
echo "  - restart your Mac"

echo
echo "=== Final values ==="
defaults read NSGlobalDomain NSNavPanelExpandedStateForSaveMode 2>/dev/null || echo "NSNavPanelExpandedStateForSaveMode = <missing>"
defaults read NSGlobalDomain NSNavPanelExpandedStateForSaveMode2 2>/dev/null || echo "NSNavPanelExpandedStateForSaveMode2 = <missing>"
defaults read NSGlobalDomain AppleShowScrollBars 2>/dev/null || echo "AppleShowScrollBars = <missing>"
defaults read NSGlobalDomain AppleScrollerPagingBehavior 2>/dev/null || echo "AppleScrollerPagingBehavior = <missing>"
defaults read NSGlobalDomain com.apple.mouse.tapBehavior 2>/dev/null || echo "com.apple.mouse.tapBehavior = <missing>"
defaults read NSGlobalDomain NSAutomaticCapitalizationEnabled 2>/dev/null || echo "NSAutomaticCapitalizationEnabled = <missing>"
defaults read NSGlobalDomain ApplePressAndHoldEnabled 2>/dev/null || echo "ApplePressAndHoldEnabled = <missing>"
defaults read NSGlobalDomain KeyRepeat 2>/dev/null || echo "KeyRepeat = <missing>"
defaults read NSGlobalDomain InitialKeyRepeat 2>/dev/null || echo "InitialKeyRepeat = <missing>"
defaults read NSGlobalDomain AppleKeyboardUIMode 2>/dev/null || echo "AppleKeyboardUIMode = <missing>"
defaults read NSGlobalDomain com.apple.keyboard.fnState 2>/dev/null || echo "com.apple.keyboard.fnState = <missing>"
```
