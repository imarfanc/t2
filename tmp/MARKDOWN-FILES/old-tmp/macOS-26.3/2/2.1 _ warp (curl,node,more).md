## check versions . py

- If you are pasting this into Warp/zsh directly

```sh
uv run --with rich python3 - <<'PY'
# script here
PY
```

```sh
uv run --with rich python3 - <<'PY'
from __future__ import annotations

import shutil
import subprocess
from pathlib import Path

from rich import box
from rich.console import Console
from rich.table import Table
from rich.text import Text

console = Console()


def run(cmd: list[str]) -> tuple[int, str]:
    try:
        result = subprocess.run(
            cmd,
            check=False,
            capture_output=True,
            text=True,
        )
        output = (result.stdout or result.stderr or "").strip()
        return result.returncode, (output if output else f"exit {result.returncode}")
    except FileNotFoundError:
        return 127, "not found"


def version_for(name: str) -> str:
    if name == "nvm":
        nvm_sh = Path.home() / ".nvm" / "nvm.sh"
        if not nvm_sh.exists():
            return "not found"
        code, output = run(
            [
                "bash",
                "-lc",
                f'. "{nvm_sh}" >/dev/null 2>&1 && '
                f'nvm --version && printf "\\ncurrent: " && nvm current',
            ]
        )
        return output if code == 0 else output

    commands = {
        "node": ["node", "--version"],
        "npm": ["npm", "--version"],
        "cargo": ["cargo", "--version"],
        "rustc": ["rustc", "--version"],
        "bun": ["bun", "--version"],
        "deno": ["deno", "--version"],
        "uv": ["uv", "--version"],
        "opencode": ["opencode", "--version"],
        "claude": ["claude", "--version"],
        "kimi": ["kimi", "--version"],
        "gemini": ["gemini", "--version"],
        "kilo": ["kilo", "--version"],
        "codex": ["codex", "--version"],
        "brew": ["brew", "--version"],
        "xcode-select": ["xcode-select", "-p"],
    }

    cmd = commands.get(name)
    if cmd is None:
        return "not supported"

    code, output = run(cmd)

    if name == "xcode-select" and code != 0:
        return "not found"

    return output


def which_for(command_name: str) -> str:
    if command_name == "nvm":
        p = Path.home() / ".nvm" / "nvm.sh"
        return str(p) if p.exists() else "not found"
    return shutil.which(command_name) or "not found"


def color_text(value: str, ok_style: str, bad_style: str) -> Text:
    style = (
        bad_style if value == "not found" or value.startswith("not found") else ok_style
    )
    return Text(value, style=style)


tools = [
    ("🟢 nvm", "nvm"),
    ("🟩 node", "node"),
    ("📦 npm", "npm"),
    ("🦀 cargo", "cargo"),
    ("⚙️  rustc", "rustc"),
    ("🥟 bun", "bun"),
    ("🦕 deno", "deno"),
    ("🚀 uv", "uv"),
    ("🔓 opencode", "opencode"),
    ("🤖 claude", "claude"),
    ("🌙 kimi", "kimi"),
    ("✨ gemini", "gemini"),
    ("🧪 kilo", "kilo"),
    ("🧠 codex", "codex"),
    ("🍺 brew", "brew"),
    ("📍 xcode-select", "xcode-select"),
]

line_after = {"npm", "uv", "codex"}

title = Text("🧰 Developer Tool Versions", style="bold magenta")
table = Table(
    title=title,
    box=box.ROUNDED,
    show_lines=False,
    header_style="bold cyan",
    expand=True,
    padding=(0, 1),
)

table.add_column("Tool", style="bold white", width=18, no_wrap=True)
table.add_column("Version", overflow="fold", ratio=3)

for label, command_name in tools:
    version = version_for(command_name)

    version_text = color_text(version, "green", "bold red")

    end_section = command_name in line_after

    table.add_row(
        Text(label, style="bold white"),
        version_text,
        end_section=end_section,
    )

console.print()
console.print(table)
console.print(
    "[dim]Red = missing/problem • Green = version output[/dim]"
)

which_table = Table(
    title=Text("Resolved Tool Paths", style="bold yellow"),
    box=box.SIMPLE,
    header_style="bold magenta",
)
which_table.add_column("Tool", style="dim", width=20)
which_table.add_column("Path", style="yellow", overflow="fold")

for label, command_name in tools:
    which_table.add_row(label, which_for(command_name))

console.print()
console.print(which_table)
PY
```

## improved .zshrc??

```sh
# zsh config 3

# opencode
export PATH="$HOME/.opencode/bin:$PATH"

# local bin (claude, uv, kimi, etc.)
[ -f "$HOME/.local/bin/env" ] && . "$HOME/.local/bin/env"

# cargo (rust)
[ -f "$HOME/.cargo/env" ] && . "$HOME/.cargo/env"

# nvm (node)
export NVM_DIR="$HOME/.nvm"
[ -s "$NVM_DIR/nvm.sh" ] && . "$NVM_DIR/nvm.sh"
[ -s "$NVM_DIR/bash_completion" ] && . "$NVM_DIR/bash_completion"

# bun
export BUN_INSTALL="$HOME/.bun"
export PATH="$BUN_INSTALL/bin:$PATH"
[ -s "$HOME/.bun/_bun" ] && source "$HOME/.bun/_bun"

# deno
[ -f "$HOME/.deno/env" ] && . "$HOME/.deno/env"

if [[ ":$FPATH:" != *":$HOME/.zsh/completions:"* ]]; then
  export FPATH="$HOME/.zsh/completions:$FPATH"
fi
```
