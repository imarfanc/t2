```sh
# zsh config 4 - Improved version

# ============================================
# Helper Functions
# ============================================

# Safely add directories to PATH
add_to_path() {
  if [ -d "$1" ] && [[ ":$PATH:" != *":$1:"* ]]; then
    export PATH="$1:$PATH"
  fi
}

# Safely source files if they exist
safe_source() {
  [ -f "$1" ] && . "$1"
}

# ============================================
# PATH Management
# ============================================

# opencode
add_to_path "$HOME/.opencode/bin"

# local bin (claude, uv, kimi, etc.)
safe_source "$HOME/.local/bin/env"

# cargo (rust)
safe_source "$HOME/.cargo/env"

# ============================================
# JavaScript/TypeScript Runtime Managers
# ============================================

# nvm (node)
export NVM_DIR="$HOME/.nvm"
safe_source "$NVM_DIR/nvm.sh"
safe_source "$NVM_DIR/bash_completion"

# bun
export BUN_INSTALL="$HOME/.bun"
add_to_path "$BUN_INSTALL/bin"
safe_source "$HOME/.bun/_bun"

# deno
safe_source "$HOME/.deno/env"

# ============================================
# ZSH Completions
# ============================================

# Add custom completions directory to FPATH
if [[ ":$FPATH:" != *":$HOME/.zsh/completions:"* ]]; then
  export FPATH="$HOME/.zsh/completions:$FPATH"
fi

# Enable zsh completions
autoload -Uz compinit && compinit

# ============================================
# ZSH Plugins (with safety checks)
# ============================================

# zsh-autocomplete
[ -f /opt/homebrew/share/zsh-autocomplete/zsh-autocomplete.plugin.zsh ] && \
  source /opt/homebrew/share/zsh-autocomplete/zsh-autocomplete.plugin.zsh

# zsh-autosuggestions
[ -f /opt/homebrew/share/zsh-autosuggestions/zsh-autosuggestions.zsh ] && \
  source /opt/homebrew/share/zsh-autosuggestions/zsh-autosuggestions.zsh

# zsh-syntax-highlighting
[ -f /opt/homebrew/share/zsh-syntax-highlighting/zsh-syntax-highlighting.zsh ] && \
  source /opt/homebrew/share/zsh-syntax-highlighting/zsh-syntax-highlighting.zsh

# ============================================
# Optional: Additional Improvements
# ============================================

# Set default editor (optional)
export EDITOR="nano"  # or "vim", "code", etc.

# History settings (optional)
export HISTSIZE=10000
export SAVEHIST=10000
setopt HIST_IGNORE_ALL_DUPS
setopt HIST_FIND_NO_DUPS

# Auto CD (optional - type directory name without 'cd')
# setopt AUTO_CD

# ============================================
# Final PATH cleanup (remove duplicates)
# ============================================

# Remove duplicate PATH/FPATH entries
typeset -U path PATH fpath FPATH
```
