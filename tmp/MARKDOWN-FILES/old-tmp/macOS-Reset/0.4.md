### works

#### no logout required

```sh
# Expand Save Dialouge
defaults write NSGlobalDomain NSNavPanelExpandedStateForSaveMode -bool true
defaults write NSGlobalDomain NSNavPanelExpandedStateForSaveMode2 -bool true

# Show ScrollBars Always, Automatic or WhenScrolling
defaults write NSGlobalDomain AppleShowScrollBars -string "WhenScrolling"
# ScrollBars-jump to the spot that's clicked
defaults write NSGlobalDomain AppleScrollerPagingBehavior -bool true

# click to tap
defaults write NSGlobalDomain "com.apple.mouse.tapBehavior" -int 0

# capitalize words automaitaclly
defaults write NSGlobalDomain NSAutomaticCapitalizationEnabled -bool false
# Disable press-and-hold for accents globally
defaults write -g ApplePressAndHoldEnabled -bool false
```

#### must logout

```sh
defaults write -g ApplePressAndHoldEnabled -bool false

# keyboard KeyRepeat rate
defaults write NSGlobalDomain KeyRepeat -int 2
# keyboard delay until KeyRepeat rate
defaults write NSGlobalDomain InitialKeyRepeat -int 15
# keyboard navigation
defaults write NSGlobalDomain AppleKeyboardUIMode -int 3
# enable function keys
defaults write NSGlobalDomain "com.apple.keyboard.fnState" -bool true

osascript -e 'tell application "System Events" to log out'
```
